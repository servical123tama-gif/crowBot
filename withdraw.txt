│                                                                              │
│ def get_capster_earnings(capster_name, aliases, start_date, end_date,        │
│ commission_rate):                                                            │
│     # 1. Ambil transaksi dari sheets (sudah ada via report_service)          │
│     df = sheets.get_transactions_by_range(start_date, end_date)              │
│                                                                              │
│     # 2. Filter by capster (termasuk alias)                                  │
│     all_names = {capster_name.lower()}                                       │
│     for alias in aliases:                                                    │
│         all_names.add(alias.lower())                                         │
│     capster_df = df[df['Capster'].str.lower().isin(all_names)]               │
│                                                                              │
│     # 3. Hitung                                                              │
│     total_revenue = capster_df['Price'].sum()                                │
│     commission = total_revenue * commission_rate  # e.g. 50%                 │
│                                                                              │
│     return {                                                                 │
│         'total_revenue': total_revenue,                                      │
│         'commission': commission,                                            │
│         'transaction_count': len(capster_df)                                 │
│     }                                                                        │
│                                                                              │
│ Periode default:                                                             │
│ - Minggu ini: Senin-Minggu minggu berjalan                                   │
│ - Opsi: bisa pilih minggu lalu atau bulan ini                                │
